#hello.py
from flask import Flask

app = Flask(__name__)

@app.route("/")
def hello():
    return "Hello World from Google App Engine!"

if __name__ == "__main__":
    app.run(host="127.0.0.1", port=8080, debug=True)


#app.yaml
runtime: python39
entrypoint: gunicorn -b :$PORT main:app

instance_class: F1

automatic_scaling:
  min_instances: 0
  max_instances: 1

#to run - terminal
python hello.py

#addition.py
from flask import Flask

app = Flask(__name__)

@app.route("/")
def add():
    a = 10
    b = 20
    return f"Addition = {a+b}"

if __name__ == "__main__":
    app.run(debug=True)

#run
python addition.py
