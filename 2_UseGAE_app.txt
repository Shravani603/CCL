#hello.py
from flask import Flask

app = Flask(__name__)

@app.route("/")
def hello():
    return "Hello World from Google App Engine!"

if __name__ == "__main__":
    app.run(host="127.0.0.1", port=8080, debug=True)


#app.yaml
runtime: python39
entrypoint: gunicorn -b :$PORT main:app


#requirements.txt
Flask==3.0.0
